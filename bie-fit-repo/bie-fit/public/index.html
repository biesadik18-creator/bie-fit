Arsitektur Hosting Cloudflare Pages + D1 SQLite + Cron WA

Skema database D1 (`schema.sql`) & Cron Worker (`worker.js`) siap pakai untuk deployment aplikasi Bie.Fit di Cloudflare:

-- Cloudflare D1 schema.sql
CREATE TABLE users (
  id TEXT PRIMARY KEY,
  email TEXT UNIQUE NOT NULL,
  wa_phone TEXT,
  target_calories INTEGER DEFAULT 2100,
  target_weight REAL DEFAULT 66.0
);

CREATE TABLE body_metrics (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  user_id TEXT,
  recorded_date TEXT,
  weight_kg REAL,
  body_fat_pct REAL,
  muscle_mass_kg REAL,
  visceral_fat REAL,
  front_photo_url TEXT,
  side_photo_url TEXT,
  back_photo_url TEXT
);
